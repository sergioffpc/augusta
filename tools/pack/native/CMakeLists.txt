# The cooker's native part (ADR-0030): one pybind11 module,
# augusta_pack._native, bound in module.cpp - the mesh pass over meshoptimizer
# (ADR-0016, mesh.h), the texture compression over stb_image,
# stb_image_resize2 and libktx (ADR-0017, texture.h), a Character's Skeleton and
# Clips built into archives by ozz-animation's offline libraries (ADR-0066,
# animation.h), and augusta_assets' pack
# encoder, which writes every pack the cook makes (encoder_binding.h), its
# reader, which the inspect, verify and publish tools read packs with
# (reader_binding.h), and its key generation, which augusta-keygen makes the
# signing keys with (keys_binding.h). Built by the root build with AUGUSTA_TOOLS, beside the Agents' module, and only with the
# tools environment's Python (the root CMakeLists.txt), which the vcpkg
# manifest's pack feature installs these dependencies for. The cooker's
# bootstrap builds this target alone (`make pack-native`).
#
# Neither the module nor augusta_assets links USD. The pipeline's own USD access
# goes through usd-optimize's pip-installed pxr build instead (cook.py), so that
# there is only ever one OpenUSD build loaded in the process - see ADR-0030 for
# why a second, independently-built one (e.g. vcpkg's own) can't coexist with
# it (a DLL-name collision between the two builds' same-named DLLs).
if(NOT AUGUSTA_TOOLS_PYTHON)
  return()
endif()

find_package(meshoptimizer CONFIG REQUIRED)
find_package(Ktx CONFIG REQUIRED)
find_package(Stb REQUIRED)

# Built straight into the package's source directory, so `import
# augusta_pack._native` works once the package is installed editable (uv sync
# in tools/, or the cooker's bootstrap), as the Agents' module is
# (tools/agent/native).
set(AUGUSTA_PACK_PACKAGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../src/augusta_pack")
pybind11_add_module(
  augusta_pack_py
  module.cpp
  animation.cpp
  encoder_binding.cpp
  keys_binding.cpp
  mesh.cpp
  reader_binding.cpp
  texture.cpp
)
target_link_libraries(
  augusta_pack_py
  PRIVATE augusta_assets meshoptimizer::meshoptimizer KTX::ktx ozz::animation_offline
)
target_include_directories(augusta_pack_py SYSTEM PRIVATE "${Stb_INCLUDE_DIR}")
set_target_properties(
  augusta_pack_py
  PROPERTIES
    OUTPUT_NAME "_native"
    RUNTIME_OUTPUT_DIRECTORY "${AUGUSTA_PACK_PACKAGE_DIR}"
    LIBRARY_OUTPUT_DIRECTORY "${AUGUSTA_PACK_PACKAGE_DIR}"
)

# The protocol's Python, augusta_pack.augusta_pb2, generated by vcpkg's protoc
# from the one schema the protocol module's C++ is generated from (ADR-0007),
# into the package beside the native module and never checked in. A part of the
# module's target, so the cooker's bootstrap (`make pack-native`) makes it too.
# augusta-inspect reads captures with it through the protobuf runtime, whose
# major matches protoc's (../pyproject.toml).
set(AUGUSTA_PROTOCOL_PROTO_DIR "${PROJECT_SOURCE_DIR}/src/modules/protocol/proto")
protobuf_generate(
  TARGET augusta_pack_py
  LANGUAGE python
  PROTOS "${AUGUSTA_PROTOCOL_PROTO_DIR}/augusta.proto"
  IMPORT_DIRS "${AUGUSTA_PROTOCOL_PROTO_DIR}"
  PROTOC_OUT_DIR "${AUGUSTA_PACK_PACKAGE_DIR}"
)
